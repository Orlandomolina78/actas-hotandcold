/**
 * Hot and Cold · Actas — envío de correo desde Gmail de la empresa.
 *
 * Este script se despliega como "aplicación web" en la cuenta
 * hycsystem176746@gmail.com. La app del celular le manda el PDF y el script
 * lo envía siempre a esa misma cuenta, más el segundo destinatario (el cliente).
 *
 * Cómo publicarlo: vea DEPLOY_APPS_SCRIPT.md (pasos en español).
 *
 * Propiedades del script (Archivo / Configuración del proyecto / Propiedades):
 *   TOKEN  = el mismo secreto que se pega en Ajustes de la app (mín. 24 caracteres)
 *
 * La app llama con POST y Content-Type text/plain (JSON en el cuerpo),
 * para evitar el "preflight" CORS que Apps Script no atiende bien.
 * Respuesta JSON: {ok:true, id:"ENT-2026-001", to:[...]} o {ok:false, error:"..."}.
 * Si el mismo acta_id llega otra vez (reintento de la cola), no se vuelve a enviar.
 */
var COMPANY_EMAIL = 'hycsystem176746@gmail.com';
var COMPANY_NAME  = 'HOT ANDA COLD SYSTEM SAS';
var BRAND         = 'Hot and Cold';
var MAX_PDF_BYTES = 15 * 1024 * 1024;
var SENT_KEY      = 'SENT_IDS';
var SENT_MAX      = 400;

function doGet() {
  return json_({
    ok: true,
    service: 'actas-hotandcold',
    hint: 'Use POST (text/plain) para enviar un acta o para ping.'
  });
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(20000);
  } catch (err) {
    return json_({ ok: false, error: 'El script está ocupado. Se reintentará.' });
  }
  try {
    return handlePost_(e);
  } catch (err) {
    return json_({ ok: false, error: String(err && err.message ? err.message : err) });
  } finally {
    try { lock.releaseLock(); } catch (e2) {}
  }
}

function handlePost_(e) {
  var raw = (e && e.postData && e.postData.contents) ? e.postData.contents : '';
  var data;
  try {
    data = JSON.parse(raw);
  } catch (err) {
    return json_({ ok: false, error: 'JSON inválido' });
  }
  if (!data || typeof data !== 'object') {
    return json_({ ok: false, error: 'JSON inválido' });
  }

  var props = PropertiesService.getScriptProperties();
  var expected = String(props.getProperty('TOKEN') || '');
  if (!expected || expected.length < 24 || expected === 'CAMBIE-ESTE-TOKEN') {
    return json_({ ok: false, error: 'Token del script no configurado' });
  }
  if (String(data.token || '') !== expected) {
    Utilities.sleep(400);
    return json_({ ok: false, error: 'Token inválido' });
  }

  if (data.ping === 1 || data.ping === '1' || data.ping === true) {
    return json_({ ok: true, service: 'actas-hotandcold' });
  }

  var actaId = String(data.acta_id || '').trim();
  if (!/^((RET|REC|ENT|CC)-\d{4}-\d{3,6}|OS-\d{1,7})$/.test(actaId)) {
    return json_({ ok: false, error: 'Consecutivo inválido' });
  }

  var sent = readSent_(props);
  if (sent.indexOf(actaId) !== -1) {
    return json_({ ok: true, id: actaId, duplicate: true, to: [COMPANY_EMAIL] });
  }

  var pdfBytes;
  try {
    var b64 = String(data.pdf_base64 || '').replace(/\s/g, '');
    var comma = b64.indexOf(',');
    if (b64.indexOf('base64') !== -1 && comma !== -1) b64 = b64.slice(comma + 1);
    if (!b64) return json_({ ok: false, error: 'No llegó el PDF' });
    pdfBytes = Utilities.base64Decode(b64);
  } catch (err) {
    return json_({ ok: false, error: 'PDF inválido (base64)' });
  }
  if (!pdfBytes || pdfBytes.length < 8) {
    return json_({ ok: false, error: 'No llegó el PDF' });
  }
  if (pdfBytes.length > MAX_PDF_BYTES) {
    return json_({ ok: false, error: 'El PDF supera el tamaño permitido (15 MB)' });
  }
  if (!(pdfBytes[0] == 37 && pdfBytes[1] == 80 && pdfBytes[2] == 68 && pdfBytes[3] == 70 && pdfBytes[4] == 45)) {
    return json_({ ok: false, error: 'El archivo no es un PDF' });
  }

  var tipoRaw = String(data.tipo || '').toLowerCase();
  var tipo = (tipoRaw === 'retiro' || tipoRaw === 'recepcion' || tipoRaw === 'recepción') ? 'retiro'
    : (tipoRaw === 'orden' || tipoRaw === 'cobro') ? tipoRaw : 'entrega';
  var toExtra = normalizeEmail_(data.to_extra);
  var ccList = parseEmailList_(data.cc);
  var to = [COMPANY_EMAIL];
  if (toExtra && toExtra !== COMPANY_EMAIL.toLowerCase()) {
    to.push(toExtra);
  }

  var filename = sanitizeFilename_(data.filename || (actaId + '.pdf'));
  var subject = String(data.subject || defaultSubject_(tipo, actaId)).slice(0, 200);
  var body = String(data.body || '').slice(0, 8000);
  if (!body) body = defaultSubject_(tipo, actaId) + '\n\nAdjuntamos el acta firmada en sitio.\n\n' + COMPANY_NAME;

  var blob = Utilities.newBlob(pdfBytes, 'application/pdf', filename);

  try {
    GmailApp.sendEmail(to.join(','), subject, body, {
      htmlBody: textToHtml_(body),
      attachments: [blob],
      name: BRAND + ' · Actas',
      cc: ccList.join(','),
      replyTo: COMPANY_EMAIL
    });
  } catch (err) {
    return json_({ ok: false, error: 'No se pudo enviar el correo. Se reintentará.' });
  }

  sent.push(actaId);
  if (sent.length > SENT_MAX) sent = sent.slice(-SENT_MAX);
  props.setProperty(SENT_KEY, JSON.stringify(sent));

  return json_({ ok: true, id: actaId, to: to });
}

function readSent_(props) {
  try {
    var raw = props.getProperty(SENT_KEY) || '[]';
    var arr = JSON.parse(raw);
    return (arr && arr.length) ? arr.slice() : [];
  } catch (err) {
    return [];
  }
}

function normalizeEmail_(v) {
  var e = String(v || '').trim().toLowerCase();
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(e)) return '';
  return e;
}

function parseEmailList_(v) {
  if (!v) return [];
  var parts = Array.isArray(v) ? v : String(v).split(/[,;]+/);
  var out = [];
  for (var i = 0; i < parts.length; i++) {
    var e = normalizeEmail_(parts[i]);
    if (e && e !== COMPANY_EMAIL.toLowerCase() && out.indexOf(e) === -1) out.push(e);
  }
  return out;
}

function sanitizeFilename_(name) {
  var n = String(name || 'acta.pdf').replace(/[^A-Za-z0-9._-]+/g, '_');
  if (!/\.pdf$/i.test(n)) n += '.pdf';
  return n.slice(0, 120);
}

function defaultSubject_(tipo, id) {
  var t = tipo === 'retiro' ? 'Acta de recepción' : tipo === 'orden' ? 'Orden de servicio' : tipo === 'cobro' ? 'Cuenta de cobro' : 'Acta de entrega';
  return t + ' ' + id + ' · ' + BRAND;
}

function textToHtml_(text) {
  var esc = String(text)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;');
  var body = esc.replace(/\n/g, '<br>');
  return '<div style="font-family:Arial,Helvetica,sans-serif;color:#1c2430;max-width:640px;font-size:14px;line-height:1.45">'
    + '<div style="background:#1B365D;color:#fff;padding:14px 18px;border-bottom:4px solid #0C6B6B">'
    + '<b style="font-size:18px">' + COMPANY_NAME + '</b><br>'
    + '<span style="font-size:13px">' + BRAND + ' · Acta de equipos</span></div>'
    + '<div style="padding:16px 4px">' + body + '</div>'
    + '<p style="font-size:13px;color:#555">' + COMPANY_NAME + ' · NIT 901.694.443-5 · 311 866 1394 · ' + COMPANY_EMAIL + '</p>'
    + '</div>';
}

function json_(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
